if BUMP_PREFIX in keys(distro.conf):
    PREFIX = value(BUMP_PREFIX)
else:
    PREFIX = first_letters( value(DISTRIBUTION) )        # Asgard -> asg
RELEASE = value(VERSION)                                 # R, e.g. 1 (thor 1)

# Pass 1:
For each Package in selected:
    
    Source = Package.Source
    sidecar(Source).rebuild = no
    sidecar(Source).built = no

    # pristine (epoch kept)
    # e.g 1:1.2.3-4+deb12u4+b2 -> version = 1:1.2.3-4+deb12u4, binNMU = +b2
    # e.g 1:1.2.3-4+b1 -> version = 1:1.2.3-4, binNMU = +b1
    # e.g 1:1.2.3-4+deb12u4 -> version = 1:1.2.3-4+deb12u4, binNMU = empty
    # e.g 1:1.2.3-4~bpo1+M -> version = 1:1.2.3-4, binNMU = ~bpo1+M
    version = Package.version.strip_binNMU
    binNMU = Package.getbinNMU                  # return +bN or ~bpoN+M if present

    if Package is tunneled:
        For each binary in Source:
            if binary not on disk:
                download binary by binary.upstream_filename
        continue

    # first time coming across the package
    if Source not in sidecar:
        sidecar(Source).rebuild = yes
        sidecar(Source).bN_BumpCount = 0

        if exists Source.patch_set:
            sidecar(Source).patch_set = current_patch_set
            sidecar(Source).Patch_Bump_Count = 1
        else
            sidecar(Source).Patch_Bump_Count = 0
            sidecar(Source).patch_set = empty
    
    # Source version changed, doesnt matter if its base or nmu
    # especially doing > , will handle downgrades seperately
    else if Source.version > sidecar(Source).previous_version: 
        sidecar(Source).rebuild = yes
        
        # we had own patches for the same version
        # patchs have to match verion including NMU since
        # since we have source changes withing different NMU
        if exists Source.patch_set:
            sidecar(Source).Patch_Bump_Count = 1
        else
            sidecar(Source).Patch_Bump_Count = 0

    else if Source.patch_set changed:
        sidecar(Source).rebuild = yes
        if exists Source.patch_set:
            sidecar(Source).Patch_Bump_Count++
        else 
            sidecar(Source).Patch_Bump_Count = 0

    # its a binNMU change, we dont need to rebuild
    # will handle backports and binary only rebuilds seperately
    else if binNMU != empty
        sidecar(Source).rebuild = no
        sidecar(Source).bN_BumpCount = 0

# Pass 2:
For each Package in selected:

    Source = Package.Source
    
    # check if package needs to be rebuild
    if sidecar(Source).rebuild == yes:
        # mark Source done for all other packages it generates
        sidecar(Source).rebuild = no
        build_source(Source, tunneled = no, force = no)

    else if Package is tunneled:
        if sidecar(Source).built == no:
            build_source(Source, tunneled = yes, force = no)


function transpose (version)
    # only the trailing debNuN part is changed. Any other suffix is retained
    # leading ~/+ is not changed 
    replace(version, trailing <debNuN>, <PREFIX><RELEASE>u<N>)
    return version

# when we need to force build a package
# where we ourself want to bump the package vide +binNMU
# force applies to all binaries of Package(Source)
function force_build(Package)
    Source = Package.Source
    build_source(Source, tunneled = no, force = yes)


function build_source(Source, tunneled, force)

    if sidecar(Source).built == yes:
        return
    
    if tunneled == false:
        Build Source(with patches)
    
        if force = yes:
            sidecar(Source).bN_BumpCount++
        else
            sidecar(Source).bN_BumpCount = 0
        
    for each binary in Source
        base_version = binary.version
        binary.X-upstream-version = (Package in selected).version

        # converts the debian syntax to asg syntax
        # e.g. 1.2.3-4+deb12u3 -> 1.2.3-4+asg1u3
        transposed_version = transpose(binary.version)

        if sidecar(Source).Patch_Bump_Count > 0:
            N = sidecar(Source).Patch_Bump_Count
            transposed_version = transposed_version + +p<N>

        if force = yes:
            N = sidecar(Source).bN_BumpCount
            transposed_version = transposed_version + +b<N>

        # sequence of application transpose → +p<N> → +b<N>
        binary.version = transposed_version

        # this is broad enough and should cover all cases
        # source builds as such shouldnt generate +bN
        for each dep in binary:
            dep_version = dep.version
            
            if dep.version has nmu:
                dep_version = transpose(dep_version)
            
            # special case where dependency is on one of the binaries generated from the same source 
            # we are force building so we have to add +bN to the dep too
            if dep in Source.binaries:

                if force == yes:
                    N = sidecar(Source).bN_BumpCount
                    dep_version = dep_version + +b<N>
                    
            dep.version = dep_version

        sign binary

    # saving state to sidecar
    sidecar(Source).previous_version = Source.version    
    sidecar(Source).built = yes
    sidecar(Source).patch_set = current_patch_set
